Pada tugas praktik mandiri ini, Anda akan merancang dan mengimplementasikan endpoint **CRUD (*Create, Read, Update, Delete*)** lengkap untuk sistem inventaris produk, lengkap dengan fitur **Pencarian (Search), Filter Kategori, Pengurutan (Sorting), serta Pagination berbasis Limit & Offset**.

---

## 1. Spesifikasi Teknis Endpoint

Endpoint yang harus diimplementasikan:

| Method | Endpoint | Fungsi | Status Sukses |
| :--- | :--- | :--- | :--- |
| `GET` | `/api/v1/products` | Mengambil daftar produk (dukung `page`, `limit`, `category`, `search`, `sortBy`) | `200 OK` |
| `POST` | `/api/v1/products` | Membuat produk baru (validasi `name`, `price`, `stock`, `category`) | `201 Created` |
| `GET` | `/api/v1/products/:id` | Mengambil detail 1 produk berdasarkan ID | `200 OK` (atau `404 Not Found`) |
| `PUT` | `/api/v1/products/:id` | Memperbarui seluruh field produk | `200 OK` |
| `DELETE` | `/api/v1/products/:id` | Menghapus produk | `200 OK` atau `204 No Content` |

### Format Standar JSON Response Envelope:
```json
{
  "success": true,
  "statusCode": 200,
  "message": "Daftar produk berhasil dimuat",
  "data": [ ... ],
  "meta": {
    "page": 1,
    "limit": 10,
    "totalItems": 45,
    "totalPages": 5,
    "hasNextPage": true,
    "hasPrevPage": false
  }
}
```

// 1. READ ALL (dengan Filter, Search, Sort, dan Pagination)
function getProducts({ page = 1, limit = 2, category, search, sortBy = "id", order = "asc" }) {
  let filtered = [...productsDB];

  // Filter Kategori
  if (category) {
    filtered = filtered.filter(p => p.category.toLowerCase() === category.toLowerCase());
  }

  // Filter Pencarian Nama
  if (search) {
    filtered = filtered.filter(p => p.name.toLowerCase().includes(search.toLowerCase()));
  }

  // Sorting
  filtered.sort((a, b) => {
    if (order === "desc") {
      return a[sortBy] < b[sortBy] ? 1 : -1;
    }
    return a[sortBy] > b[sortBy] ? 1 : -1;
  });

  // Pagination Calculation
  const totalItems = filtered.length;
  const totalPages = Math.ceil(totalItems / limit) || 1;
  const validPage = Math.max(1, Math.min(page, totalPages));
  const offset = (validPage - 1) * limit;
  const paginatedData = filtered.slice(offset, offset + limit);

  return {
    success: true,
    statusCode: 200,
    data: paginatedData,
    meta: {
      page: validPage,
      limit,
      totalItems,
      totalPages,
      hasNextPage: validPage < totalPages,
      hasPrevPage: validPage > 1
    }
  };
}

// 2. CREATE PRODUCT
function createProduct(payload) {
  if (!payload.name || typeof payload.price !== "number" || !payload.category) {
    return {
      success: false,
      statusCode: 422,
      error: "Validasi Gagal: 'name', 'price' (number), dan 'category' wajib diisi"
    };
  }

  const newId = productsDB.length > 0 ? Math.max(...productsDB.map(p => p.id)) + 1 : 1;
  const newProduct = {
    id: newId,
    name: payload.name,
    price: payload.price,
    category: payload.category,
    stock: payload.stock || 0,
    createdAt: new Date().toISOString().split("T")[0]
  };

  productsDB.push(newProduct);

  return {
    success: true,
    statusCode: 201,
    message: "Produk berhasil ditambahkan",
    data: newProduct
  };
}

// 3. DELETE PRODUCT
function deleteProduct(id) {
  const index = productsDB.findIndex(p => p.id === id);
  if (index === -1) {
    return { success: false, statusCode: 404, error: \`Produk dengan ID \${id} tidak ditemukan\` };
  }

  const deleted = productsDB.splice(index, 1)[0];
  return { success: true, statusCode: 200, message: \`Produk '\${deleted.name}' berhasil dihapus\` };
}

## 3. Checklist Penyelesaian Tugas Mandiri

Sebelum menandai materi ini selesai, pastikan Anda memahami:
- [ ] Formula kalkulasi *Offset*: `offset = (page - 1) * limit`.
- [ ] Penggunaan *Sanitization* query parameter (mengonversi string `"2"` menjadi integer numerik).
- [ ] Perbedaan antara *Offset-based Pagination* (cocok untuk dashboard tabel) vs *Cursor-based Pagination* (cocok untuk Infinite Scroll seperti feed Twitter/Instagram).
