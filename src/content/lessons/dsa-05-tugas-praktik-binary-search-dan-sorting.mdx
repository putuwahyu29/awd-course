## 1. Spesifikasi Tugas yang Harus Diimplementasikan

Diberikan sekumpulan objek transaksi penjualan:
1. **Fungsi `sortTransactionsByPriceDesc`**: Mengurutkan ribuan data transaksi dari harga tertinggi ke terendah secara efisien (`O(N log N)`).
2. **Fungsi `findTransactionById`**: Melakukan pencarian transaksi berdasarkan ID unik dalam waktu logaritmik `O(log N)` menggunakan Binary Search.
3. **Fungsi `countTransactionsInRange(sortedArr, minPrice, maxPrice)`**: Menghitung berapa banyak transaksi yang berada di dalam rentang harga menggunakan dua titik Binary Search (*Lower Bound* & *Upper Bound*).

function sortTransactionsByAmount(arr) {
  if (arr.length <= 1) return arr;

  const pivot = arr[Math.floor(arr.length / 2)];
  const left = [];
  const middle = [];
  const right = [];

  for (let i = 0; i < arr.length; i++) {
    if (arr[i].amount < pivot.amount) {

    } else if (arr[i].amount > pivot.amount) {
      right.push(arr[i]);
    } else {
      middle.push(arr[i]);
    }
  }

  return [...sortTransactionsByAmount(left), ...middle, ...sortTransactionsByAmount(right)];
}

// 2. Binary Search Berdasarkan ID Transaksi (Array harus terurut berdasarkan ID)
function findTransactionById(sortedByIdArr, targetId) {
  let left = 0;
  let right = sortedByIdArr.length - 1;

  while (left <= right) {
    const mid = Math.floor((left + right) / 2);
    if (sortedByIdArr[mid].id === targetId) {
      return sortedByIdArr[mid];
    }
    if (sortedByIdArr[mid].id < targetId) {
      left = mid + 1;
    } else {
      right = mid - 1;
    }
  }

  return null;
}
