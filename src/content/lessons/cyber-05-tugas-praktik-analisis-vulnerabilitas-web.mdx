Pada tugas praktik mandiri ini, Anda bertindak sebagai **Application Security Engineer (AppSec)** yang ditugaskan untuk mengaudit potongan kode backend dan frontend yang rentan, mengidentifikasi 3 jenis celah keamanan fatal (IDOR, SQL Injection, dan XSS), lalu menulis kode *patch* perbaikan.

---

## 1. Spesifikasi Kode Rentan yang Harus Dibenahi

Perhatikan potongan kode backend Node.js berikut:

```javascript
// POTONGAN KODE RENTAN YANG HARUS DI-AUDIT:
app.get('/api/users/:id/profile', async (req, res) => {
  // Masalah 1: IDOR (Tidak memeriksa apakah req.params.id sama dengan req.user.id)
  const userId = req.params.id;
  
  // Masalah 2: SQL Injection (Penggabungan string mentah ke query)
  const query = "SELECT * FROM users WHERE id = '" + userId + "'";
  const [user] = await db.query(query);
  
  // Masalah 3: XSS / Information Disclosure (Mengembalikan password hash ke client)
  res.send(`<h1>Profil Pengguna: ${user.name}</h1><p>Bio: ${user.bio}</p>`);
});
```

function escapeHtml(unsafeText) {
  if (typeof unsafeText !== "string") return "";
  return unsafeText
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")

// Controller yang Telah Di-Patch (Aman)
function secureGetProfileHandler(req) {
  const requestedUserId = req.params.id;
  const currentLoggedInUser = req.user; // Dari JWT Guard terverifikasi

  // 1. PATCH IDOR: Pastikan user hanya dapat mengakses datanya sendiri (atau jika role ADMIN)
  if (currentLoggedInUser.role !== "ADMIN" && currentLoggedInUser.id !== requestedUserId) {
    return {
      statusCode: 403,
      body: { success: false, error: "Forbidden: Anda tidak berhak melihat profil pengguna lain" }
    };
  }

  // 2. PATCH SQL INJECTION: Parameterized Lookup (bukan string concatenation)
  const user = mockDatabase.find(u => u.id === requestedUserId);
  if (!user) {
    return { statusCode: 404, body: { success: false, error: "User tidak ditemukan" } };
  }

  // 3. PATCH XSS & SENSITIVE DATA EXPOSURE:
  // - Hilangkan passwordHash
  // - Sanitasi string bio sebelum dirender
  const safeProfile = {
    id: user.id,
    name: escapeHtml(user.name),
    bio: escapeHtml(user.bio) // Tag <script> dinetralkan menjadi &lt;script&gt;
  };

  return {
    statusCode: 200,
    body: { success: true, data: safeProfile }
  };
}
